    return t.split("|").map(function (c) { return c.trim(); });
  }

  function markdown(src) { return markdownLines(esc(src || "").replace(/\r/g, "").split("\n")); }

  // Строки уже экранированы: блок ```details размечается рекурсивно из тех же строк.
  function markdownLines(lines) {
    var out = [], i = 0, para = [];
    function flush() { if (para.length) { out.push("<p>" + inline(para.join("<br>")) + "</p>"); para = []; } }
    while (i < lines.length) {
      var line = lines[i];
      var fence = /^```\s*([a-zA-Z]*)/.exec(line);
      if (fence) {
        flush(); var code = []; i++;
        while (i < lines.length && !/^```/.test(lines[i])) { code.push(lines[i]); i++; }
        var lang = (fence[1] || "").toLowerCase();
        // Текст уже экранирован; служебные блоки отдаются обработчикам через textContent.
        if (lang === "mermaid") out.push("<pre class='diagram-pending hidden'>" + code.join("\n") + "</pre>");
        else if (lang === "suggest") out.push("<pre class='suggest-pending hidden'>" + code.join("\n") + "</pre>");
        else if (lang === "details") out.push("<div class='details'><a class='more' href='#'></a><div>" + markdownLines(code) + "</div></div>");
        else out.push("<pre><code>" + code.join("\n") + "</code></pre>");
        i++; continue;
      }
      var h = /^(#{1,3})\s+(.*)$/.exec(line);
      if (h) { flush(); out.push("<h" + h[1].length + ">" + inline(h[2]) + "</h" + h[1].length + ">"); i++; continue; }
      if (/\|/.test(line) && i + 1 < lines.length && isTableSep(lines[i + 1])) {
        flush(); var head = cells(line); i += 2; var rows = [];
        while (i < lines.length && /\|/.test(lines[i]) && lines[i].trim() !== "") { rows.push(cells(lines[i])); i++; }
        var t = "<table class='data'><tr>" + head.map(function (c) { return "<th>" + inline(c) + "</th>"; }).join("") + "</tr>";
        rows.forEach(function (r) { t += "<tr>" + r.map(function (c) { return "<td>" + inline(c) + "</td>"; }).join("") + "</tr>"; });
        out.push(t + "</table>"); continue;
      }
      if (/^\s*[-*]\s+/.test(line)) {
        flush(); var items = [];
        while (i < lines.length && /^\s*[-*]\s+/.test(lines[i])) { items.push("<li>" + inline(lines[i].replace(/^\s*[-*]\s+/, "")) + "</li>"); i++; }
        out.push("<ul>" + items.join("") + "</ul>"); continue;
      }
      if (/^\s*\d+[.)]\s+/.test(line)) {
        flush(); var oitems = [];
        while (i < lines.length && /^\s*\d+[.)]\s+/.test(lines[i])) { oitems.push("<li>" + inline(lines[i].replace(/^\s*\d+[.)]\s+/, "")) + "</li>"); i++; }
        out.push("<ol>" + oitems.join("") + "</ol>"); continue;
      }
      if (line.trim() === "") { flush(); i++; continue; }
      para.push(line); i++;
    }
    flush();
    return out.join("");
  }

  // Автоссылки в тексте реплики: точные вхождения представлений объектов хода (от длинных к коротким).
  function linkObjects(root, objects) {
    var list = (objects || []).filter(function (o) { return o && o.id != null && o.text && String(o.text).length >= 4; })
      .sort(function (a, b) { return String(b.text).length - String(a.text).length; });
    if (!list.length) return;
    var walker = document.createTreeWalker(root, NodeFilter.SHOW_TEXT, null, false);
    var nodes = [], n;
    while ((n = walker.nextNode())) {
      if (n.parentNode && n.parentNode.nodeName !== "A" && n.parentNode.nodeName !== "CODE") nodes.push(n);
    }
    nodes.forEach(function (node) {
      var text = node.nodeValue, parts = [], pos = 0, changed = false;
      while (pos < text.length) {
        var best = null, bestAt = -1;
        list.forEach(function (o) {
          var at = text.indexOf(String(o.text), pos);
          if (at >= 0 && (bestAt < 0 || at < bestAt)) { best = o; bestAt = at; }
        });
        if (!best) break;
        if (bestAt > pos) parts.push(document.createTextNode(text.slice(pos, bestAt)));
        parts.push(objLink(best.id, String(best.text)));
        pos = bestAt + String(best.text).length;
        changed = true;
      }
      if (!changed) return;
      if (pos < text.length) parts.push(document.createTextNode(text.slice(pos)));
      var frag = document.createDocumentFragment();
      parts.forEach(function (x) { frag.appendChild(x); });
      node.parentNode.replaceChild(frag, node);
    });
  }

  function fmtNum(v) {
    var a = Math.abs(v), r;
    if (a >= 1e9) r = (v / 1e9).toFixed(1) + " млрд";
    else if (a >= 1e6) r = (v / 1e6).toFixed(1) + " млн";
    else if (a >= 1e4) r = (v / 1e3).toFixed(0) + " тыс";
    else r = String(Math.round(v * 100) / 100);
    return r.replace(".", ",");
  }

  function drawAxes(holder, labels, series, visible, type) {
    var hbar = type === "hbar", W = 680, rowH = 22;
    var H = hbar ? Math.max(120, labels.length * rowH + 30) : 300;
    var padL = hbar ? 170 : 56, padB = hbar ? 22 : 60, padT = 10, padR = 16;
    var svg = svgEl("svg", { viewBox: "0 0 " + W + " " + H, role: "img" });
    var vals = [];
    series.forEach(function (s, i) { if (visible[i]) vals = vals.concat(s.values); });
    var max = Math.max.apply(null, vals.concat([0])), min = Math.min.apply(null, vals.concat([0]));
    if (max === min) max = min + 1;
    var plotW = W - padL - padR, plotH = H - padT - padB;
    function scale(v) { return (v - min) / (max - min); }
    // Сетка и подписи шкалы.
    for (var g = 0; g <= 4; g++) {
      var gv = min + (max - min) * g / 4;
      if (hbar) {
        var gx = padL + plotW * g / 4;
        svgEl("line", { x1: gx, y1: padT, x2: gx, y2: H - padB, stroke: "#eaeef2" }, svg);
        svgEl("text", { x: gx, y: H - 6, "font-size": 10, "text-anchor": "middle", fill: "#5f6b7a" }, svg).textContent = fmtNum(gv);
      } else {
        var gy = padT + plotH - plotH * g / 4;
        svgEl("line", { x1: padL, y1: gy, x2: W - padR, y2: gy, stroke: "#eaeef2" }, svg);
        svgEl("text", { x: padL - 6, y: gy + 3, "font-size": 10, "text-anchor": "end", fill: "#5f6b7a" }, svg).textContent = fmtNum(gv);
      }
    }
    var shown = [];
    series.forEach(function (s, i) { if (visible[i]) shown.push(i); });
    var n = labels.length, band = (hbar ? plotH : plotW) / Math.max(1, n);
    if (type === "line") {
      shown.forEach(function (si) {
        var pts = series[si].values.map(function (v, k) {
          return [padL + band * (k + 0.5), padT + plotH - plotH * scale(v)];
        });
        svgEl("polyline", { points: pts.map(function (p) { return p.join(","); }).join(" "), fill: "none",
          stroke: PALETTE[si % PALETTE.length], "stroke-width": 2 }, svg);
        pts.forEach(function (p, k) {
          var c = svgEl("circle", { cx: p[0], cy: p[1], r: 3.5, fill: PALETTE[si % PALETTE.length], "class": "bar" }, svg);
          tip(c, labels[k] + " — " + series[si].name + ": " + fullNum(series[si].values[k]));
        });
      });
    } else {
      var inner = band * 0.8 / Math.max(1, shown.length);
      shown.forEach(function (si, j) {
        series[si].values.forEach(function (v, k) {
          var r;
          if (hbar) {
            var y = padT + band * k + band * 0.1 + inner * j;
            var x0 = padL + plotW * scale(Math.min(0, v)), w = Math.abs(plotW * (scale(v) - scale(0)));
            r = svgEl("rect", { x: x0, y: y, width: Math.max(1, w), height: inner, fill: PALETTE[si % PALETTE.length], "class": "bar" }, svg);
          } else {
            var x = padL + band * k + band * 0.1 + inner * j;
            var y1 = padT + plotH - plotH * scale(Math.max(0, v)), h = Math.abs(plotH * (scale(v) - scale(0)));
            r = svgEl("rect", { x: x, y: y1, width: Math.max(1, inner - 1), height: Math.max(1, h), fill: PALETTE[si % PALETTE.length], "class": "bar" }, svg);
          }
          tip(r, labels[k] + " — " + series[si].name + ": " + fullNum(v));
        });
      });
    }
    // Подписи категорий.
    labels.forEach(function (l, k) {
      if (hbar) {
        svgEl("text", { x: padL - 6, y: padT + band * (k + 0.5) + 3, "font-size": 11, "text-anchor": "end", fill: "#1f2328" }, svg).textContent = shortLabel(l, 28);
      } else if (n <= 24 || k % Math.ceil(n / 24) === 0) {
        var tx = padL + band * (k + 0.5), ty = H - padB + 12;
        var te = svgEl("text", { x: tx, y: ty, "font-size": 10, "text-anchor": "end", fill: "#1f2328",
          transform: "rotate(-35 " + tx + " " + ty + ")" }, svg);
        te.textContent = shortLabel(l, 18);
      }
    });
    holder.appendChild(svg);
  }

  function drawPie(holder, labels, s) {
    var W = 300, R = 120, cx = 150, cy = 140;
    var svg = svgEl("svg", { viewBox: "0 0 " + W + " 280", role: "img", style: "max-width:320px" });
    var vals = (s ? s.values : []).map(function (v) { return Math.max(0, v); });
    var total = vals.reduce(function (a, b) { return a + b; }, 0) || 1, a0 = -Math.PI / 2;
    vals.forEach(function (v, k) {
      var a1 = a0 + 2 * Math.PI * v / total, large = a1 - a0 > Math.PI ? 1 : 0;
      var d = vals.length === 1 ? "M" + (cx - R) + "," + cy + "a" + R + "," + R + " 0 1,0 " + 2 * R + ",0a" + R + "," + R + " 0 1,0 " + (-2 * R) + ",0"
        : "M" + cx + "," + cy + "L" + (cx + R * Math.cos(a0)) + "," + (cy + R * Math.sin(a0)) +
          "A" + R + "," + R + " 0 " + large + ",1 " + (cx + R * Math.cos(a1)) + "," + (cy + R * Math.sin(a1)) + "Z";
      var p = svgEl("path", { d: d, fill: PALETTE[k % PALETTE.length], stroke: "#fff", "class": "slice" }, svg);
      tip(p, labels[k] + ": " + fullNum(v) + " (" + Math.round(v * 1000 / total) / 10 + "%)");
      a0 = a1;
    });
    holder.appendChild(svg);
  }

  function makeSortable(table, rowsData) {
    var head = table.rows[0];
    Array.prototype.forEach.call(head.cells, function (th, col) {
      th.className = "sortable";
      var asc = true;
      th.onclick = function () {
        var body = Array.prototype.slice.call(table.rows, 1);
        body.sort(function (a, b) {
          var x = a.cells[col].getAttribute("data-v"), y = b.cells[col].getAttribute("data-v");
          var nx = parseFloat(x), ny = parseFloat(y);
          var r = (!isNaN(nx) && !isNaN(ny) && String(nx) === x && String(ny) === y) ? nx - ny : String(x).localeCompare(String(y), "ru");
          return asc ? r : -r;
        });
        asc = !asc;
        body.forEach(function (tr) { table.appendChild(tr); });
      };
    });
  }

  // ---- Источники: сноски [k] в тексте и список под ответом. Ссылки - только по номеру реестра формы ----
  function linkCites(root, byN) {
    var walker = document.createTreeWalker(root, NodeFilter.SHOW_TEXT, null, false);
    var nodes = [], n;
    while ((n = walker.nextNode())) {
      var up = n.parentNode;
      if (up && up.nodeName !== "A" && up.nodeName !== "CODE" && up.nodeName !== "PRE" && /\[\d{1,3}\]/.test(n.nodeValue)) nodes.push(n);
    }
    nodes.forEach(function (node) {
      var re = /\[(\d{1,3})\]/g, text = node.nodeValue, pos = 0, m, frag = document.createDocumentFragment();
      while ((m = re.exec(text))) {
        var src = byN[Number(m[1])];
        if (!src) continue;
        if (m.index > pos) frag.appendChild(document.createTextNode(text.slice(pos, m.index)));
        var sup = document.createElement("sup"); sup.className = "cite";
        if (src.id !== null && src.id !== undefined) sup.appendChild(objLink(src.id, "[" + Number(m[1]) + "]", "cite"));
        else sup.textContent = "[" + Number(m[1]) + "]";
        sup.title = String(src.text || "");
        frag.appendChild(sup);
        pos = m.index + m[0].length;
      }
      if (!pos) return;
      if (pos < text.length) frag.appendChild(document.createTextNode(text.slice(pos)));
      node.parentNode.replaceChild(frag, node);
    });
  }

  // ---- «Открыть в Excel» у таблиц из текста ответа: данные уходят в 1С каналом data-cmd ----
  var XL_ROWS = 2000, XL_COLS = 50, XL_CELL = 1000, XL_JSON = 2 * 1024 * 1024;
  function cellText(td) {
    var c = td.cloneNode(true);
    Array.prototype.forEach.call(c.querySelectorAll("sup.cite"), function (x) { x.parentNode.removeChild(x); });
    return String(c.textContent || "").replace(/^\s+|\s+$/g, "");
  }
  function questionOf(div) {
    var x = div.previousSibling;
    while (x && !(x.getAttribute && x.getAttribute("data-q") !== null)) x = x.previousSibling;
    return x ? x.getAttribute("data-q") : "";
  }
  function tableCmd(table, title) {
    var rows = Array.prototype.slice.call(table.rows);
    if (!rows.length || rows.length - 1 > XL_ROWS) return null;
    var columns = Array.prototype.map.call(rows[0].cells, cellText);
    if (columns.length > XL_COLS) return null;
    var data = [], tooLong = false;
    rows.slice(1).forEach(function (tr) {
      var r = Array.prototype.map.call(tr.cells, cellText).slice(0, XL_COLS);
      r.forEach(function (v) { if (v.length > XL_CELL) tooLong = true; });
      data.push(r);
    });
    columns.forEach(function (v) { if (v.length > XL_CELL) tooLong = true; });
    if (tooLong) return null;
    var json = JSON.stringify({ cmd: "excel", title: String(title || "").slice(0, 150), columns: columns, rows: data });
    return json.length > XL_JSON ? null : json;
  }
  function addExcelButtons(body, div) {
    Array.prototype.forEach.call(body.querySelectorAll("table.data"), function (table) {
      var bar = document.createElement("div"); bar.className = "tbl-bar";
      var a = document.createElement("a"); a.className = "more tbl-tools"; a.href = "#iia/cmd"; a.textContent = "Открыть в Excel";
      a.addEventListener("click", function (e) {
        var json = tableCmd(table, questionOf(div));
        if (!json) { e.preventDefault(); e.stopPropagation(); setStatus("Таблица слишком большая для выгрузки."); return; }
        document.getElementById("iia-request").setAttribute("data-cmd", json);
      }, true);
      bar.appendChild(a);
      table.parentNode.insertBefore(bar, table.nextSibling);
    });
  }

  // Подготовка скриншота. Снимок нескольких мониторов (PrtScr на 2-3 экрана) режется на части примерно 16:9 -
  // иначе при уменьшении до размера, который принимает модель, текст становится нечитаемым.
  // Каждая часть: не больше 1920 px по длинной стороне, JPEG с качеством от 0,85 до 0,5 - пока не уложится в ~900 КБ.
  var MAX_SIDE = 1920, MAX_LEN = 1200000, SPLIT_RATIO = 2.2;
  // Ограничения исходного снимка: больше - отказ, а не попытка ужать до нечитаемого.
  var SRC_MAX_BYTES = 20 * 1024 * 1024, SRC_MAX_PIXELS = 25000000, SRC_MAX_WIDTH = 11520, SRC_MAX_HEIGHT = 4320;

  function encodePart(img, sx, sy, sw, sh) {
    var k = Math.min(1, MAX_SIDE / Math.max(sw, sh));
    var c = document.createElement("canvas");
    c.width = Math.max(1, Math.round(sw * k)); c.height = Math.max(1, Math.round(sh * k));
    var g = c.getContext("2d");
    g.fillStyle = "#fff"; g.fillRect(0, 0, c.width, c.height);
    g.imageSmoothingEnabled = true;
    g.drawImage(img, sx, sy, sw, sh, 0, 0, c.width, c.height);
    var q = 0.85, url = c.toDataURL("image/jpeg", q);
    while (url.length > MAX_LEN && q > 0.5) { q -= 0.1; url = c.toDataURL("image/jpeg", q); }
    return url.length > MAX_LEN ? null : url;
  }

  function canAddImage(size) {
    if (!config.images) { hint.textContent = "Вставка изображений отключена администратором."; return false; }
    if (!config.imageModel) { hint.textContent = "Скриншот не добавлен: ни одна доступная модель не принимает изображения. Администратору: флажок «Принимает изображения» в подключении."; return false; }
    if (pending.length >= config.maxImages) { hint.textContent = "Не больше " + config.maxImages + " изображений в сообщении."; return false; }
    if (size > SRC_MAX_BYTES) {
      hint.textContent = "Файл слишком большой (" + Math.round(size / 1048576) + " МБ, допустимо до 20 МБ). Вставьте снимок окна (Alt+PrtScr) или области (Win+Shift+S).";
      return false;
    }
    return true;
  }

  function addImageFile(file) {

  }

  function addImageUrl(src) {
      var img = new Image();
      img.onerror = function () { hint.textContent = "Не удалось прочитать изображение."; };
      img.onload = function () {
        var w = img.width, h = img.height, parts = 1;
        if (w * h > SRC_MAX_PIXELS || w > SRC_MAX_WIDTH || h > SRC_MAX_HEIGHT) {
          hint.textContent = "Снимок слишком большой (" + w + "x" + h + " px; допустимо до 11520x4320 и 25 Мпикс). Вставьте снимок окна (Alt+PrtScr) или области (Win+Shift+S).";
          return;
        }
        if (w / h > SPLIT_RATIO) parts = Math.max(2, Math.round((w / h) / (16 / 9)));
        if (pending.length + parts > config.maxImages) {
          hint.textContent = "Снимок на " + parts + " экрана не помещается: не больше " + config.maxImages + " изображений в сообщении.";
          return;
        }
        var step = Math.ceil(w / parts), added = 0;
        for (var i = 0; i < parts; i++) {
          var sx = i * step, sw = Math.min(step, w - sx);
          var url = encodePart(img, sx, 0, sw, h);
          if (url) { pending.push(url); added++; }
        }
        renderAttachments();
        if (!added) hint.textContent = "Изображение слишком большое.";
        else if (parts > 1) hint.textContent = "Широкий снимок разделён на " + parts + " части - по экранам. Скриншоты не проверяются на персональные данные.";
      };
      img.src = src;
  }

  // Статус хода с таймером: сколько идёт текущий шаг и весь ход - видно, что агент работает, а не завис.
  var statusText = "", stepSince = 0, turnSince = 0;
  function mmss(ms) { var s = Math.floor(ms / 1000); return Math.floor(s / 60) + ":" + ("0" + (s % 60)).slice(-2); }
  function renderStatus() {
    if (!busy || !statusText) { statusLine.textContent = statusText; return; }
    var now = Date.now(), text = statusText + "... " + mmss(now - stepSince);
    if (stepSince - turnSince > 1000) text += " (всего " + mmss(now - turnSince) + ")";
    statusLine.textContent = text;
  }
  function setStatus(text) {
    text = (text || "").replace(/\.\.\.$/, "");
    if (text !== statusText) { statusText = text; stepSince = Date.now(); }
    renderStatus();
  }

  var UUID_RE = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i;
  function queuedNode(id) { return UUID_RE.test(id || "") ? document.getElementById("q-" + id) : null; }
  function renderQueueState(div, id, held, reason) {

    renderQueueState(div, p.id, !!p.held, p.reason || "");
    queueBox.appendChild(div);
    queueBox.scrollTop = queueBox.scrollHeight;
  }
  function queueHeld(p) {
    var div = queuedNode(p.id);
    if (div) renderQueueState(div, p.id, !!p.held, p.reason || "");
  }
  function unqueued(p) {
    var div = queuedNode(p.id);
    if (div) div.parentNode.removeChild(div);
  }
  // Текст отменённого или не принятого сообщения возвращается в поле ввода, не затирая набранное.
  function restoreInput(p) {
    var text = typeof p.text === "string" ? p.text : "";
    if (text) input.value = input.value.trim() ? input.value + "\n" + text : text;
    outbox.value = ""; outbox.removeAttribute("data-pending");
    if (p.hint) hint.textContent = p.hint;
    input.focus();
  }

  function copyMessage(id) {
    var el = document.getElementById("m" + id);
    if (!el) return;
    copyText(el.getAttribute("data-text") || "");
  }

  // Мост 1С → JS: единственная точка входа, пакет — JSON-строка.
  window.iiaReceive = function (json) {
    var p;
    try { p = JSON.parse(json); } catch (e) { return "bad-json"; }
    if (p.type === "append") append(p);
    else if (p.type === "status") setStatus(p.text);
    else if (p.type === "progress") liveStep(p);
    else if (p.type === "dialogs") showDialogs(p);
    else if (p.type === "replace") {
      var old = feed.querySelector(".msg[data-msg='" + Math.max(0, Math.floor(Number(p.msg) || 0)) + "']");
      if (old) old.parentNode.removeChild(old);
    }
    else if (p.type === "queued") queued(p);
    else if (p.type === "unqueued") unqueued(p);
    else if (p.type === "queue-held") queueHeld(p);
    else if (p.type === "restore-input") restoreInput(p);
    else if (p.type === "clear") { feed.innerHTML = ""; queueBox.innerHTML = ""; live.textContent = ""; live.className = "hidden"; setStatus(""); lastQuestion = ""; input.value = ""; pending = []; sentShots = []; pendingDocs = []; pendingBase = []; outbox.value = ""; outbox.removeAttribute("data-pending"); renderAttachments(); }
    else if (p.type === "capabilities") showCapabilities(p);
    else if (p.type === "config") {
      config.images = !!p.images; config.imageModel = p.imageModel !== false; if (p.maxImages) config.maxImages = p.maxImages;
      config.docs = !!p.docs; config.textDocs = !!p.textDocs; config.baseFiles = !!p.baseFiles; if (p.maxDocs) config.maxDocs = p.maxDocs;
      config.models = p.models || []; config.model = p.model || "";
      // Состояние панели истории - только при открытии чата; на узком окне панель поверх ленты, сначала скрыта.
      if (typeof p.sidebar === "boolean" && !sideKnown) { sideKnown = true; setSide(p.sidebar && !narrow(), false); }
      // Без модели с документами убираются только PDF; без разрешения файлов - все файлы.
      var before = pendingDocs.length + pendingBase.length;
      pendingDocs = pendingDocs.filter(function (d) { return config.textDocs && (config.docs || !PDF_EXT.test(d.name)); });
      if (!config.textDocs) pendingBase = [];
      var dropped = pendingDocs.length + pendingBase.length < before;
      renderModels(); renderAttachments();
      if (dropped) hint.textContent = config.textDocs ? "Выбранная модель не принимает PDF - PDF-файлы убраны." : "Файлы в вопросах отключены - файлы убраны.";
    }
    else if (p.type === "busy") setBusy(p.value);
    else if (p.type === "accepted") {
      // Сообщение принято 1С: очищаем ввод, вложения покажем в ленте вместе с вопросом.
      lastQuestion = input.value; sentShots = pending.slice(0); pending = []; pendingDocs = []; pendingBase = []; input.value = ""; outbox.value = ""; outbox.removeAttribute("data-pending"); renderAttachments();
    }
    else if (p.type === "focus") input.focus();
    else if (p.type === "change") updateChange(p.card);
    return "ok";
  };

  // Копирование обрабатывается в JS; остальные команды #iia/... уходят в 1С через событие ПриНажатии.
  document.addEventListener("click", function (e) {
    var a = e.target && e.target.closest ? e.target.closest("a") : null;
    if (!a) return;
    var href = a.getAttribute("href") || "";
    var m = /^#iia\/copy\/(\d+)$/.exec(href);
    if (m) { copyMessage(m[1]); }
    // Кнопки карточки изменения блокируются до ответа 1С (повторный клик не уходит; сервер всё равно защищён).
    if (/^#iia\/change\/\d+\/(exec|reject)$/.test(href) && a.parentNode && a.parentNode.classList) {
      if (a.parentNode.classList.contains("busy")) { e.preventDefault(); e.stopPropagation(); return; }
      setTimeout(function () { a.parentNode.classList.add("busy"); }, 0);
    }
  }, true);

  window.iiaReady = true;
})();
</script>
